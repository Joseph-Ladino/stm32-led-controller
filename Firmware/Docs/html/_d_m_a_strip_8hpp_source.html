98 inline void onDMAInterrupt(bool halfCompleteInterrupt) {
99
100 static DMABufferType *leftHalf = dmaBuffer, *rightHalf = dmaBuffer + halfDmaBufferSize;
101 static constexpr uint16_t bitsPerBufferElement = StripType::bitsPerPixel * sizeof(DMABufferType);
102
103 DMABufferType *writeBuf = halfCompleteInterrupt ? (DMABufferType*) leftHalf : (DMABufferType*) rightHalf;
104
105 uint16_t curPixel;
106 for (curPixel = 0; curPixel < numPixelsPerDMA && dmaLedIT != dmaLedEndIT; dmaLedIT++, curPixel++) {
107 auto led = *dmaLedIT;
108 colorToDMATiming(led, &writeBuf);
109 }
110
111 // buffer has been completely set to reset value, dma can be stopped
112 if (dmaTransferStatus == DMAStatus::RESET_COMPLETE) {
113 dmaTransferStatus = DMAStatus::DONE;
114 dmaStop();
115 // frameDelay.reset();
116
117 return;
118 }
119
120 // all data in ledstrip has been written but the DMA buffer is not full of reset data
121 if (curPixel != numPixelsPerDMA) {
122 auto fillPixels = numPixelsPerDMA - curPixel;
123
124 // fill with reset value to transition to WS28xx reset period
125 memset(writeBuf, 0, (uint32_t) fillPixels * bitsPerBufferElement);
126
127
128 if (fillPixels == numPixelsPerDMA && dmaTransferStatus != DMAStatus::IN_PROGRESS) {
129 dmaTransferStatus =
130 (dmaTransferStatus == DMAStatus::RESET_STARTED) ?
131 DMAStatus::RESET_IN_PROGRESS : DMAStatus::RESET_COMPLETE;
132 } else {
133 dmaTransferStatus = DMAStatus::RESET_STARTED;
134 }
135 }
136 }
