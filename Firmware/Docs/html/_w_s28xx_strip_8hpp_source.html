237 inline void display() {
238 if (!state.powerOn) return;
239
240 // TODO: filters/effects/animations
241 static Color colCrt = state.colorCorrection, tmpCrt = state.tempCorrection;
242
243 uint32_t adjust[4];
244 for (int i = 0; i < 4; i++) {
245 adjust[i] = ((((uint32_t) colCrt[i]) + 1) * (((uint32_t) tmpCrt[i]) + 1) * state.brightness) >> 16;
246 }
247
248 // at the expense of 256 extra bytes of memory, this solution felt cleaner
249 const uint8_t *gammaLookupArr = state.gammaEnabled ? gammaCorrectionLookup : gammaRawLookup;
250
251 for (uint16_t i = 0; i < numPixels; i++) {
252 Color pixel = rawBuffer[i];
253 for (uint8_t j = 0; j < 4; j++) {
254 pixel[j] = ((uint32_t) gammaLookupArr[pixel[j]] * adjust[j]) >> 8;
255 }
256
257 fxBuffer[i] = StripType::getOrderedBytes(pixel);
258 }
259 }
